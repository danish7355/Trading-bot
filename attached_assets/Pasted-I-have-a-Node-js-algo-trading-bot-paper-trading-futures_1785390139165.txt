Create a new file: services/exitManager.js

It must monitor all open positions on every candle close and evaluate these

EXIT_TRIGGERS = [
  "hard_stop_hit",           // Price crossed stop level
  "regime_deterioration",    // Regime score dropped below 40
  "momentum_exhaustion",     // RSI divergence on 3+ consecutive candles
  "trend_structure_break",   // Price closed below EMA_M after being above it
  "volume_dry_up",           // 3 consecutive candles below 0.5× volume SMA
  "max_hold_time_exceeded",  // Position held beyond timeframe profile limit
  "funding_rate_extreme",    // Funding rate flipped extreme against position direction
  "oi_dropping",             // OI dropped >10% while price moved in direction (exhaustion)
  "manual_exit",             // User clicked exit in UI or sent Telegram command
  "kill_switch",             // Emergency exit all — triggered by kill switch above
]

Exit decision logic (translate to Node.js exactly):

async function evaluateExit(position, marketState) {
  const triggersFired = []

  // Hard stop — always checked first, exits immediately, never batched
  if (position.side === 'LONG' && marketState.price <= position.stopPrice) {
    return { exit: true, reason: 'HARD_STOP', pctToExit: 100 }
  }
  if (position.side === 'SHORT' && marketState.price >= position.stopPrice) {
    return { exit: true, reason: 'HARD_STOP', pctToExit: 100 }
  }

  // Regime deterioration
  if (marketState.regime.score < 40) triggersFired.push('REGIME_DETERIORATION')

  // Trend structure break
  if (position.side === 'LONG' && marketState.close < marketState.emaM) {
    triggersFired.push('TREND_STRUCTURE_BREAK')
  }
  if (position.side === 'SHORT' && marketState.close > marketState.emaM) {
    triggersFired.push('TREND_STRUCTURE_BREAK')
  }

  // RSI divergence — price making higher high but RSI making lower high
  if (detectRsiDivergence(position, marketState.df)) {
    triggersFired.push('MOMENTUM_EXHAUSTION')
  }

  // Volume dry-up — 3 consecutive candles below 0.5× volume SMA
  if (marketState.volRatio3Period < 0.5) triggersFired.push('VOLUME_DRY_UP')

  // Max hold time
  if (new Date() > position.maxHoldUntil) triggersFired.push('MAX_HOLD_TIME')

  // Funding rate extreme
  if (Math.abs(marketState.fundingRate) > FUNDING_EXTREME_THRESHOLD &&
      isFundingAgainstPosition(position.side, marketState.fundingRate)) {
    triggersFired.push('FUNDING_RATE_EXTREME')
  }

  // OI dropping while price moves in direction (exhaustion signal)
  if (marketState.oiDropPct > 0.10) triggersFired.push('OI_DROPPING')

  // Exit decision matrix — DO NOT change this logic:
  // 3+ soft triggers  → exit 100%
  // 2 soft triggers   → exit 50%
  // 1 soft trigger    → tighten stop to 0.8× ATR distance, don't exit
  // 0 triggers        → hold

  if (triggersFired.length >= 3) {
    return { exit: true, reason: triggersFired.join('|'), pctToExit: 100 }
  } else if (triggersFired.length === 2) {
    return { exit: true, reason: triggersFired.join('|'), pctToExit: 50 }
  } else if (triggersFired.length === 1) {
    return { exit: false, action: 'TIGHTEN_STOP', reason: triggersFired[0] }
  } else {
    return { exit: false, action: 'HOLD' }
  }
}

Additional Exit Manager requirements:
- Partial exit (pctToExit: 50): close half the position, update the DB 
  record with the partial close, recalculate and update stop/TP on the 
  remaining half. Don't create a duplicate position row — update the 
  existing one with quantity reduced and a partial_exit_log JSON field.
- TIGHTEN_STOP action: recalculate stop as current_price - (0.8 × ATR) 
  for LONG, current_price + (0.8 × ATR) for SHORT. Update position record. 
  Log the stop adjustment.
- Funding rate threshold (FUNDING_EXTREME_THRESHOLD) and max hold time 
  limits per timeframe should be configurable in settings.
- Every exit decision — even HOLD — should be logged to an 
  exit_evaluations table (position_id, timestamp, triggers_fired, decision, 
  action). This is the audit trail for strategy performance.
- manual_exit: wire the "Exit" button in the UI and a Telegram command 
  (/exit SYMBOL) to set a manual_exit_requested flag on the position. 
  The evaluateExit loop picks it up on the next cycle.
- kill_switch exit: when the kill switch is activated, call evaluateExit 
  with a synthetic marketState that forces immediate 100% exit on every 
  open position.

Telegram messages on exit:
- Send a Telegram alert for every exit with: symbol, side, exit trigger(s),

- For partial exits: clearly mark as "PARTIAL EXIT (50%)" with remaining
